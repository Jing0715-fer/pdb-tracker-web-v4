    93→async function esearch(query: string, date: string, windowDays: number, maxResults: number): Promise<string[]> {
    94→  // ±N day window — matches the real workflow's "±3 天" filter.
    95→  const center = new Date(date + 'T00:00:00Z');
    96→  const start = new Date(center);
    97→  start.setUTCDate(start.getUTCDate() - windowDays);
    98→  const end = new Date(center);
    99→  end.setUTCDate(end.getUTCDate() + windowDays);
   100→
   101→  const fmt = (d: Date) => `${d.getUTCFullYear()}/${String(d.getUTCMonth() + 1).padStart(2, '0')}/${String(d.getUTCDate()).padStart(2, '0')}`;
   102→
   103→  const url = new URL(`${EUTILS}/esearch.fcgi`);
   104→  url.searchParams.set('db', 'pubmed');
   105→  url.searchParams.set('term', query);
   106→  url.searchParams.set('mindate', fmt(start));
   107→  url.searchParams.set('maxdate', fmt(end));
   108→  url.searchParams.set('datetype', 'pdat');
   109→  url.searchParams.set('retmax', String(maxResults));
   110→  url.searchParams.set('retmode', 'json');
   111→  url.searchParams.set('sort', 'relevance');
   112→
   113→  const res = await fetch(url.toString(), {
   114→    headers: { 'User-Agent': 'pdb-tracker-web-v3/1.0' },
   115→    signal: AbortSignal.timeout(30_000),
   116→  });
   117→  if (!res.ok) throw new Error(`eSearch ${res.status}: ${await res.text().catch(() => '')}`);
   118→  const json: any = await res.json();
   119→  return json?.esearchresult?.idlist || [];
   120→}

   134→function parseArticle(xml: string): FetchedPaper | null {
   135→  const pmid = xml.match(/<PMID[^>]*>([\s\S]*?)<\/PMID>/)?.[1]?.trim();
   136→  if (!pmid) return null;
   137→
   138→  const titleMatch = xml.match(/<ArticleTitle[^>]*>([\s\S]*?)<\/ArticleTitle>/);
   139→  const title = titleMatch?.[1]?.replace(/<[^>]+>/g, '').trim() || '';
   140→
   141→  const authorBlock = xml.match(/<AuthorList[^>]*>([\s\S]*?)<\/AuthorList>/)?.[1] || '';
   142→  const authorRe = /<Author[^>]*>([\s\S]*?)<\/Author>/g;
   143→  const authors: string[] = [];
   144→  let m: RegExpExecArray | null;
   145→  while ((m = authorRe.exec(authorBlock))) {
   146→    const inner = m[1];
   147→    const collective = inner.match(/<CollectiveName[^>]*>([\s\S]*?)<\/CollectiveName>/)?.[1];
   148→    if (collective) {
   149→      authors.push(collective.replace(/<[^>]+>/g, '').trim());
   150→      continue;
   151→    }
   152→    const last = inner.match(/<LastName[^>]*>([\s\S]*?)<\/LastName>/)?.[1]?.trim() || '';
   153→    const fore = inner.match(/<ForeName[^>]*>([\s\S]*?)<\/ForeName>/)?.[1]?.trim() || '';
   154→    const full = `${fore} ${last}`.trim();
   155→    if (full) authors.push(full);
   156→  }
   157→
   158→  const journal = xml.match(/<Journal>[\s\S]*?<Title[^>]*>([\s\S]*?)<\/Title>/)?.[1]?.trim() || '';
   159→
   160→  const abstractMatch = xml.match(/<Abstract[^>]*>([\s\S]*?)<\/Abstract>/);
   161→  let abstract = '';
   162→  if (abstractMatch) {
   163→    const parts: string[] = [];
   164→    const txtRe = /<AbstractText(?:\s+Label="([^"]*)")?[^>]*>([\s\S]*?)<\/AbstractText>/g;
   165→    let mm: RegExpExecArray | null;
   166→    while ((mm = txtRe.exec(abstractMatch[1]))) {
   167→      const label = mm[1];
   168→      const text = mm[2].replace(/<[^>]+>/g, '').trim();
   169→      parts.push(label ? `${label}: ${text}` : text);
   170→    }
   171→    abstract = parts.join('\n\n');
   172→  }
   173→
   174→  const pubDateMatch =
   175→    xml.match(/<JournalIssue>[\s\S]*?<PubDate[^>]*>([\s\S]*?)<\/PubDate>/) ||
   176→    xml.match(/<ArticleDate[^>]*>([\s\S]*?)<\/ArticleDate>/) ||
   177→    xml.match(/<PubDate[^>]*>([\s\S]*?)<\/PubDate>/);
   178→  const pubXml = pubDateMatch?.[1] || '';
   179→  const year = pubXml.match(/<Year[^>]*>([\s\S]*?)<\/Year>/)?.[1]?.trim()
   180→    || pubXml.match(/<MedlineDate[^>]*>([\s\S]*?)<\/MedlineDate>/)?.[1]?.match(/(\d{4})/)?.[1]
   181→    || '';
   182→  const month = pubXml.match(/<Month[^>]*>([\s\S]*?)<\/Month>/)?.[1]?.trim() || '';
   183→  const day = pubXml.match(/<Day[^>]*>([\s\S]*?)<\/Day>/)?.[1]?.trim() || '';
   184→
   185→  const doi = xml.match(/<ArticleId[^>]*IdType="doi"[^>]*>([\s\S]*?)<\/ArticleId>/)?.[1]?.trim() || '';
   186→
   187→  return { pmid, title, authors: authors.join(', '), journal, abstract, pubYear: year, pubMonth: month, pubDay: day, doi };
   188→}

   190→async function efetch(pmids: string[]): Promise<FetchedPaper[]> {
   191→  if (pmids.length === 0) return [];
   192→  const out: FetchedPaper[] = [];
   193→  for (let i = 0; i < pmids.length; i += 50) {
   194→    const batch = pmids.slice(i, i + 50);
   195→    const url = new URL(`${EUTILS}/efetch.fcgi`);
   196→    url.searchParams.set('db', 'pubmed');
   197→    url.searchParams.set('id', batch.join(','));
   198→    url.searchParams.set('rettype', 'xml');
   199→    url.searchParams.set('retmode', 'xml');
   200→    const res = await fetch(url.toString(), {
   201→      headers: { 'User-Agent': 'pdb-tracker-web-v3/1.0' },
   202→      signal: AbortSignal.timeout(60_000),
   203→    });
   204→    if (!res.ok) throw new Error(`eFetch ${res.status}: ${await res.text().catch(() => '')}`);
   205→    const xml = await res.text();
   206→    const articleChunks = xml.split('<PubmedArticle>').slice(1);
   207→    for (const chunk of articleChunks) {
   208→      const end = chunk.indexOf('</PubmedArticle>');
   209→      if (end < 0) continue;
   210→      const articleXml = '<PubmedArticle>' + chunk.slice(0, end);
   211→      const paper = parseArticle(articleXml);
   212→      if (paper) out.push(paper);
   213→    }
   214→    if (i + 50 < pmids.length) await new Promise(r => setTimeout(r, 350));
   215→  }
   216→  return out;
   217→}

   223→async function upsertPapers(papers: FetchedPaper[]): Promise<{ inserted: number; updated: number }> {
   224→  let inserted = 0, updated = 0;
   225→  for (const p of papers) {
   226→    const titleJson = JSON.stringify(p.title);
   227→    const authorsJson = JSON.stringify(p.authors);
   228→    const abstractJson = JSON.stringify(p.abstract);
   229→
   230→    await db.$executeRaw`
   231→      INSERT INTO PubMedArticle (pubmedId, title, authors, journal, pubYear, pubMonth, pubDay, abstract, createdAt)
   232→      VALUES (${p.pmid}, ${titleJson}, ${authorsJson}, ${p.journal}, ${p.pubYear}, ${p.pubMonth}, ${p.pubDay}, ${abstractJson}, CURRENT_TIMESTAMP)
   233→      ON CONFLICT(pubmedId) DO UPDATE SET
   234→        title = excluded.title,
   235→        authors = excluded.authors,
   236→        journal = excluded.journal,
   237→        pubYear = COALESCE(NULLIF(excluded.pubYear, ''), PubMedArticle.pubYear),
   238→        pubMonth = COALESCE(NULLIF(excluded.pubMonth, ''), PubMedArticle.pubMonth),
   239→        pubDay = COALESCE(NULLIF(excluded.pubDay, ''), PubMedArticle.pubDay),
   240→        abstract = COALESCE(NULLIF(excluded.abstract, ''), PubMedArticle.abstract)
   241→    `;
   242→    // The DB has no `source` column in the production schema (despite
   243→    // prisma advertising it). We still attempt the UPDATE and silently
   244→    // ignore if the column doesn't exist.
   245→    try {
   246→      await db.$executeRawUnsafe(`UPDATE PubMedArticle SET source = ? WHERE pubmedId = ?`, '结构生物学文献日报', p.pmid);
   247→    } catch {/* column may not exist; ignore */}
   248→
   249→    const row = await db.$queryRawUnsafe<any[]>(`SELECT createdAt FROM PubMedArticle WHERE pubmedId = ?`, p.pmid);
   250→    const isFresh = row[0] && row[0].createdAt && (Date.now() - new Date(row[0].createdAt).getTime()) < 5000;
   251→    if (isFresh) inserted++; else updated++;
   252→  }
   253→  return { inserted, updated };
   254→}

   265→async function generateSummaries(papers: FetchedPaper[], llmCfg?: LlmConfig): Promise<SummarizedPaper[]> {
   266→  const out: SummarizedPaper[] = [];
   267→  // Process in small batches of 5 to keep prompts manageable. The LLM
   268→  // returns one summary per line (in the same order).
   269→  const BATCH = 5;
   270→  for (let i = 0; i < papers.length; i += BATCH) {
   271→    const batch = papers.slice(i, i + BATCH);
   272→    const lines = batch.map((p, idx) => `[${idx + 1}] PMID:${p.pmid}\n标题: ${p.title}\n期刊: ${p.journal}\n摘要: ${(p.abstract || '').slice(0, 1200)}`).join('\n\n');
   273→
   274→    const system = 'You are a structural biology expert who writes concise Chinese research summaries (中文研究概要) for daily literature digests. Each summary should be 2-4 sentences, focus on the structural mechanism / methodology, and avoid hedging language. Output exactly N lines, each starting with the paper number in brackets, no extra commentary.';
   275→    const prompt = `请为下面 ${batch.length} 篇结构生物学论文分别撰写中文研究概要。每篇 2-4 句话，重点说明：①研究问题 ②关键结构/方法发现 ③创新点。
   276→
   277→${lines}
   278→
   279→输出格式（每篇一行，不要其他内容）：
   280→[1] 中文概要
   281→[2] 中文概要
   282→...`;
   283→
   284→    const r = await llmComplete(prompt, { ...llmCfg, system });
   285→    const summaries: string[] = [];
   286→    if (r.ok && r.text) {
   287→      // Parse [N] summary lines from the LLM output.
   288→      const re = /\[(\d+)\]\s*([\s\S]*?)(?=\[\d+\]|$)/g;
   289→      let mm: RegExpExecArray | null;
   290→      while ((mm = re.exec(r.text))) {
   291→        const idx = parseInt(mm[1], 10) - 1;
   292→        summaries[idx] = mm[2].trim();
   293→      }
   294→    }
   295→
   296→    for (let j = 0; j < batch.length; j++) {
   297→      const p = batch[j];
   298→      const text = `${p.title}\n${p.abstract}`;
   299→      const method = classifyMethod(text) || 'Other';
   300→      const isHighIF = /^(Nature|Science|Cell|Nature Communications|Nature Structural|Nature Methods|Nature Biotechnology|Molecular Cell|Cell Research|Science Advances|Structure|Current Biology|eLife|PLOS Biology|Proceedings of the National Academy of Sciences)/i.test(p.journal);
   301→      out.push({
   302→        ...p,
   303→        method,
   304→        summaryCn: summaries[j] || '（摘要生成中或失败）',
   305→        doi: p.doi,
   306→        isHighIF,
   307→      });
   308→    }
   309→  }
   310→  return out;
   311→}

   335→function buildDailyReport(papers: SummarizedPaper[], date: string, windowDays: number, stats: { pathACount: number; pathBCount: number }): string {
   336→  const methodStats: Record<string, number> = {};
   337→  const journalStats: Record<string, number> = {};
   338→  for (const p of papers) {
   339→    methodStats[p.method] = (methodStats[p.method] || 0) + 1;
   340→    journalStats[p.journal] = (journalStats[p.journal] || 0) + 1;
   341→  }
   342→
   343→  const totalCandidates = stats.pathACount + stats.pathBCount;
   344→  const methodTable = Object.entries(methodStats)
   345→    .sort((a, b) => b[1] - a[1])
   346→    .map(([m, c]) => `| ${m} | ${c} |`).join('\n');
   347→  const journalTable = Object.entries(journalStats)
   348→    .sort((a, b) => b[1] - a[1])
   349→    .slice(0, 12)
   350→    .map(([j, c]) => `| ${j} | ${c} |`).join('\n');
   351→
   352→  // High-IF highlights: papers from Nature/Science/Cell family.
   353→  const highlights = papers
   354→    .filter(p => p.isHighIF)
   355→    .slice(0, 6)
   356→    .map(p => `- **[${p.journal.split(' ')[0]}]** ${p.title} (PMID: ${p.pmid}) — ${p.method}`)
   357→    .join('\n');
   358→
   359→  const paperSections = papers.map((p, idx) => {
   360→    const isHighlight = p.isHighIF && idx < 8; // stars on the first batch of high-IF
   361→    return formatPaperSection(p, idx + 1, isHighlight);
   362→  }).join('\n');
   363→
   364→  // Build a stub executive summary — this gets overwritten by LLM in stage 2.
   365→  // Note: no `## 执行摘要` heading here — the LLM stage adds its own.
   366→  const placeholderExec = `今日（${date}）筛选窗口（±${windowDays} 天）共纳入 ${papers.length} 篇结构生物学论文，方法分布：${Object.entries(methodStats).map(([m, c]) => `${m} ${c} 篇`).join('、')}。\n\n_执行摘要将由 LLM 在生成阶段填入（本节为占位文本）。_`;
   367→
   368→  return `# 结构生物学文献日报 — ${date}
   369→
   370→**报告日期**: ${date}
   371→**筛选窗口**: ±${windowDays} 天（2026-${date.slice(5, 7)} 月前后 ±${windowDays} 天浮动）
   372→**检索策略**: 双路径 PubMed 检索（MeSH+方法关键词 / 高 IF 期刊+方法关键词）
   373→**最终入选**: ${papers.length} 篇结构生物学论文
   374→
   375→## 方法分布
   376→
   377→| 方法 | 数量 |
   378→|------|------|
   379→${methodTable || '| — | 0 |'}
   380→| **合计** | **${papers.length}** |
   381→
   382→## 期刊分布（前 12）
   383→
   384→| 期刊 | 数量 |
   385→|------|------|
   386→${journalTable || '| — | 0 |'}
   387→
   388→${highlights ? `## 高亮论文（Nature/Science/Cell 系刊）\n\n${highlights}\n\n---\n\n## 论文列表\n\n` : `## 论文列表\n\n`}${paperSections}
   389→${placeholderExec}
   390→`;
   391→}

   393→function buildExecutiveSummary(papers: SummarizedPaper[], date: string, llmCfg?: LlmConfig): Promise<string> {
   394→  if (papers.length === 0) {
   395→    return Promise.resolve(`## 执行摘要\n\n今日（${date}）未检索到符合条件的结构生物学论文。`);
   396→  }
   397→  // Top 5 papers by IF / by length, give LLM enough context to write
   398→  // a thematic synthesis.
   399→  const top = papers.slice(0, 8);
   400→  const system = 'You are a structural biology expert writing a Chinese executive summary for a daily literature digest. Output ONLY the markdown section content (no surrounding headers unless needed), 3-5 paragraphs, ~500 字. Highlight thematic groupings, structural method breakthroughs, and any disease/drug relevance. Do not invent data — be precise and brief.';
   401→  const prompt = `今日（${date}）共纳入 ${papers.length} 篇结构生物学论文（按方法分布略）。请基于以下 8 篇代表论文撰写中文执行摘要（500 字，3-5 段）：
   402→
   403→${top.map((p, i) => `[${i + 1}] PMID ${p.pmid} (${p.journal}, ${p.method}): ${p.title}\n   概要: ${p.summaryCn.slice(0, 200)}`).join('\n\n')}
   404→
   405→请按主题分类撰写（不要逐篇罗列），重点强调：
   406→- 本批次最突出的结构/方法学亮点
   407→- 跨论文的方法学趋势或主题聚类
   408→- 与疾病机制或药物设计的关联（如有）
   409→
   410→直接以 ## 执行摘要 开头。`;
   411→
   412→  return llmComplete(prompt, { ...llmCfg, system }).then(r => {
   413→    if (r.ok && r.text) {
   414→      // Strip ALL `## 执行摘要` headings anywhere in the LLM output — we
   415→      // always add exactly one at the top. Hermes in particular tends to
   416→      // echo the heading inside the body, producing duplicates.
   417→      const body = r.text
   418→        .replace(/^#{1,3}\s*执行摘要\s*\n+/i, '')       // leading
   419→        .replace(/\n#{1,3}\s*执行摘要\s*\n+/g, '\n\n')  // in-body echoes
   420→        .trim();
   421→      console.log(`[lit-daily] buildExecutiveSummary body length: ${body.length}, preview: ${body.slice(0, 200).replace(/\n/g, ' ')}`);
   422→      return `## 执行摘要\n\n${body}\n\n_本报告由 pdb-tracker-web-v3 自动生成于 ${new Date().toISOString().replace('T', ' ').slice(0, 19)}_`;
   423→    }
   424→    return `## 执行摘要\n\n> ⚠️ 执行摘要生成失败: ${r.error || 'unknown'}\n\n今日（${date}）共纳入 ${papers.length} 篇结构生物学论文。详细列表见上文。\n\n_本报告由 pdb-tracker-web-v3 自动生成于 ${new Date().toISOString().replace('T', ' ').slice(0, 19)}_`;
   425→  });
   426→}

   428→// ─── Master index management ──────────────────────────────────────────────────
   429→
   430→async function updateMasterIndex(date: string): Promise<void> {
   431→  const masterPath = path.join(SB_ROOT, 'index.md');
   432→  let existing = '';
   433→  try {
   434→    existing = await fs.readFile(masterPath, 'utf-8');
   435→  } catch {/* master doesn't exist yet */}
   436→
   437→  // Parse out existing rows: lines of the form `| YYYY-MM-DD | [日报](...) |`.
   438→  const rowRe = /^\|\s*(\d{4}-\d{2}-\d{2})\s*\|\s*\[日报\]\(daily-reports\/structural-biology\/(\d{4}-\d{2}-\d{2})\/index\.md\)\s*\|\s*$/gm;
   439→  const existingDates = new Map<string, string>();
   440→  let mm: RegExpExecArray | null;
   441→  while ((mm = rowRe.exec(existing))) {
   442→    existingDates.set(mm[1], mm[2]);
   443→  }
   444→  existingDates.set(date, date);
   445→
   446→  const sorted = Array.from(existingDates.keys()).sort((a, b) => b.localeCompare(a));
   447→  const rows = sorted.map(d => `| ${d} | [日报](daily-reports/structural-biology/${d}/index.md) |`).join('\n');
   448→
   449→  const master = `# 结构生物学文献日报 - 主索引
   450→
   451→| 日期 | 链接 |
   452→|------|------|
   453→${rows}
   454→`;
   455→  await fs.mkdir(SB_ROOT, { recursive: true });
   456→  await fs.writeFile(masterPath, master, 'utf-8');
   457→}

   461→export interface LiteratureDailyOptions {
   462→  /** Target date (YYYY-MM-DD). Defaults to today. */
   463→  date?: string;
   464→  /** ±N day window. Default 3. */
   465→  windowDays?: number;
   466→  /** Per-path candidate cap. Default 400 (Path A) / 80 (Path B). */
   467→  maxPathA?: number;
   468→  maxPathB?: number;
   469→  /** Final paper cap (after dedup + filter). Default 30. */
   470→  maxPapers?: number;
   471→  /** LLM config. If absent, no per-paper summary / no executive summary. */
   472→  llm?: LlmConfig;
   473→  /** Skip writing files to the wiki (DB only). Default false (write files). */
   474→  skipWikiFiles?: boolean;
   475→  /** Optional: coarse progress callback. Receives human-readable stage
   476→   *  markers like 'esearch', 'efetch', 'persist', 'digest'. */
   477→  onStage?: (stage: string, detail?: string) => void;
   478→}

   480→export interface LiteratureDailyResult {
   481→  ok: boolean;
   482→  date: string;
   483→  windowDays: number;
   484→  pathACount: number;
   485→  pathBCount: number;
   486→  totalCandidates: number;
   487→  finalCount: number;
   488→  inserted: number;
   489→  updated: number;
   490→  methodStats: Record<string, number>;
   491→  files: {
   492→    dailyDir: string;
   493→    dailyIndex: string;
   494→    masterIndex: string;
   495→  };
   496→  durationMs: number;
   497→  error?: string;
   498→}
   499→
   500→export async function runLiteratureDaily(opts: LiteratureDailyOptions = {}): Promise<LiteratureDailyResult> {
   501→  const t0 = Date.now();
   502→  const date = opts.date || new Date().toISOString().slice(0, 10);
   503→  const windowDays = opts.windowDays ?? 3;
   504→  const maxA = opts.maxPathA ?? 400;
   505→  const maxB = opts.maxPathB ?? 80;
   506→  const maxPapers = opts.maxPapers ?? 30;
   507→  const stage = opts.onStage || (() => {});
   508→
   509→  try {
   510→    stage('init', `${date} window=±${windowDays}d maxA=${maxA} maxB=${maxB} maxPapers=${maxPapers}`);
   511→    // ─── Step 1: Two-path PubMed search ────────────────────────────────────
   512→    stage('esearch', `PubMed path A + B for ${date}`);
   513→    const [pathAIds, pathBIds] = await Promise.all([
   514→      esearch(PATH_A_QUERY, date, windowDays, maxA).catch(() => [] as string[]),
   515→      esearch(PATH_B_QUERY, date, windowDays, maxB).catch(() => [] as string[]),
   516→    ]);
   517→    stage('esearch-done', `pathA=${pathAIds.length} pathB=${pathBIds.length}`);
   518→
   519→    // Dedupe, preserve Path B priority (high-IF).
   520→    const seen = new Set<string>();
   521→    const candidates: string[] = [];
   522→    for (const id of [...pathBIds, ...pathAIds]) {
   523→      if (!seen.has(id)) { seen.add(id); candidates.push(id); }
   524→    }
   525→
   526→    if (candidates.length === 0) {
   527→      return {
   528→        ok: true, date, windowDays,
   529→        pathACount: pathAIds.length, pathBCount: pathBIds.length,
   530→        totalCandidates: 0, finalCount: 0, inserted: 0, updated: 0,
   531→        methodStats: {},
   532→        files: { dailyDir: '', dailyIndex: '', masterIndex: '' },
   533→        durationMs: Date.now() - t0,
   534→      };
   535→    }
   536→
   537→    // ─── Step 2: eFetch ────────────────────────────────────────────────────
   538→    await sleep(350);
   539→    const rawPapers = await efetch(candidates);
   540→
   541→    // ─── Step 3: Filter to structural biology methods ──────────────────────
   542→    const sbPapers = rawPapers.filter(p => {
   543→      const text = `${p.title}\n${p.abstract}`;
   544→      return classifyMethod(text) !== null;
   545→    });
   546→
   547→    // Cap to top N (sort by journal IF as a proxy for relevance).
   548→    sbPapers.sort((a, b) => {
   549→      const aJ = highIFJournals.has(a.journal) ? 1 : 0;
   550→      const bJ = highIFJournals.has(b.journal) ? 1 : 0;
   551→      if (aJ !== bJ) return bJ - aJ;
   552→      return a.title.length - b.title.length;
   553→    });
   554→    const limited = sbPapers.slice(0, maxPapers);
   555→
   556→    // ─── Step 4: Persist to DB ─────────────────────────────────────────────
   557→    stage('persist', `upserting ${limited.length} papers`);
   558→    const { inserted, updated } = await upsertPapers(limited);
   559→    stage('persist-done', `inserted=${inserted} updated=${updated}`);
   560→
   561→    // ─── Step 5: LLM summaries (per-paper Chinese summaries + exec summary) ─
   562→    stage(opts.llm ? 'llm-summarize' : 'summarize-skip', opts.llm ? `summarizing ${limited.length} papers` : 'no LLM configured');
   563→    const summarized = opts.llm
   564→      ? await generateSummaries(limited, opts.llm)
   565→      : limited.map(p => {
   566→          const text = `${p.title}\n${p.abstract}`;
   567→          const method = classifyMethod(text) || 'Other';
   568→          const isHighIF = /^(Nature|Science|Cell|Nature Communications|Nature Structural|Nature Methods|Nature Biotechnology|Molecular Cell|Cell Research|Science Advances|Structure|Current Biology|eLife|PLOS Biology|Proceedings of the National Academy of Sciences)/i.test(p.journal);
   569→          return { ...p, method, summaryCn: '（未生成中文概要，未提供 LLM 配置）', doi: p.doi, isHighIF };
   570→        });
   571→
   572→    // ─── Step 6: Markdown output ───────────────────────────────────────────
   573→    let dailyDir = '';
   574→    let dailyIndex = '';
   575→    let masterIndex = '';
   576→
   577→    if (!opts.skipWikiFiles) {
   578→      dailyDir = path.join(SB_ROOT, date);
   579→      await fs.mkdir(dailyDir, { recursive: true });
   580→      dailyIndex = path.join(dailyDir, 'index.md');
   581→      masterIndex = path.join(SB_ROOT, 'index.md');
   582→
   583→      let report = buildDailyReport(summarized, date, windowDays, { pathACount: pathAIds.length, pathBCount: pathBIds.length });
   584→
   585→      if (opts.llm) {
   586→        stage('llm-exec-summary', `executive summary for ${summarized.length} papers`);
   587→        const exec = await buildExecutiveSummary(summarized, date, opts.llm);
   588→        // The placeholder is a one-line italic stub ending with `_`. Strip it
   589→        // along with anything after it, then insert the LLM-generated
   590→        // `## 执行摘要` section.
   591→        const placeholderLine = '_执行摘要将由 LLM 在生成阶段填入（本节为占位文本）。_';
   592→        const phIdx = report.indexOf(placeholderLine);
   593→        if (phIdx >= 0) {
   594→          report = report.slice(0, phIdx).trimEnd() + '\n\n' + exec + '\n';
   595→        } else if (report.includes('## 执行摘要')) {
   596→          report = report.replace(/## 执行摘要[\s\S]*$/, exec);
   597→        } else {
   598→          report = report.trimEnd() + '\n\n' + exec + '\n';
   599→        }
   600→      }
   601→
   602→      await fs.writeFile(dailyIndex, report, 'utf-8');
   603→      await updateMasterIndex(date);
   604→    }
   605→
   606→    // ─── Step 7: Method stats for return ───────────────────────────────────
   607→    const methodStats: Record<string, number> = {};
   608→    for (const p of summarized) {
   609→      methodStats[p.method] = (methodStats[p.method] || 0) + 1;
   610→    }
   611→
   612→    return {
   613→      ok: true, date, windowDays,
   614→      pathACount: pathAIds.length,
   615→      pathBCount: pathBIds.length,
   616→      totalCandidates: candidates.length,
   617→      finalCount: summarized.length,
   618→      inserted,
   619→      updated,
   620→      methodStats,
   621→      files: { dailyDir, dailyIndex, masterIndex },
   622→      durationMs: Date.now() - t0,
   623→    };
   624→  } catch (e: any) {
   625→    return {
   626→      ok: false, date, windowDays,
   627→      pathACount: 0, pathBCount: 0, totalCandidates: 0, finalCount: 0,
   628→      inserted: 0, updated: 0, methodStats: {},
   629→      files: { dailyDir: '', dailyIndex: '', masterIndex: '' },
   630→      durationMs: Date.now() - t0,
   631→      error: e?.message || String(e),
   632→    };
   633→  }
   634→}
   635→
   636→const highIFJournals = new Set([
   637→  'Nature', 'Science', 'Cell', 'Nature Communications', 'Nature Structural & Molecular Biology',
   638→  'Nature Methods', 'Nature Biotechnology', 'Molecular Cell', 'Cell Research',
   639→  'Science Advances', 'Cell Host & Microbe', 'Structure', 'Current Biology',
   640→  'eLife', 'PLOS Biology', 'Proceedings of the National Academy of Sciences',
   641→  'Nature Medicine', 'Nature Genetics', 'Nature Immunology', 'Nature Neuroscience',
   642→  'Nature Cell Biology', 'Nature Chemical Biology',
   643→]);
